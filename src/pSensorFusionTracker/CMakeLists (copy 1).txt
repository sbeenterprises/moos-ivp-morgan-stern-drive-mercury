#==============================================================
# CMakeLists.txt for pSensorFusionTracker
# Multi-Sensor Fusion for USVs
#==============================================================

cmake_minimum_required(VERSION 3.5)
project(pSensorFusionTracker)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

#--------------------------------------------------------------
# Find MOOS
#--------------------------------------------------------------
find_package(MOOS 10.0 REQUIRED)

#--------------------------------------------------------------
# Find Eigen3 (Required for UKF)
#--------------------------------------------------------------
find_package(Eigen3 REQUIRED)

#--------------------------------------------------------------
# Include Directories
#--------------------------------------------------------------
include_directories(
    ${MOOS_INCLUDE_DIRS}
    ${EIGEN3_INCLUDE_DIR}
    .
)

#--------------------------------------------------------------
# Source Files
#--------------------------------------------------------------
set(SRC
    pSensorFusionTracker.cpp
)

#--------------------------------------------------------------
# Create Executable
#--------------------------------------------------------------
add_executable(pSensorFusionTracker ${SRC})

#--------------------------------------------------------------
# Link Libraries
#--------------------------------------------------------------
target_link_libraries(pSensorFusionTracker
    ${MOOS_LIBRARIES}
    pthread
)

#--------------------------------------------------------------
# Install
#--------------------------------------------------------------
install(TARGETS pSensorFusionTracker
    RUNTIME DESTINATION bin
)
